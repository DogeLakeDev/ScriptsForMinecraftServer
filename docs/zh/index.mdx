```ts title="sapi/src/index.ts"
import { ModuleRegistry } from "@sfmc-bds/sdk/sapi/module-loader";
import { Command, Msg, Permission } from "@sfmc-bds/sdk/sapi/runtime";

ModuleRegistry.register({
  id: "feature-hello",
  afterWorldLoad: false,
  lifecycle: {
    registerPermissions() {
      Permission.register("hello.use", "使用 hello 命令", 0);
    },
    registerCommands() {
      Command.register("hello", {
        description: "发送问候消息",
        permission: "hello.use",
        handler(player) {
          Msg.send(player, "§a你好，欢迎来到 SFMC 驱动的 Minecraft 服务器！");
        },
      });
    },
  },
});
```
